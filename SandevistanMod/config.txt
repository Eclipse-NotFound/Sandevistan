# 斯安维斯坦模组配置 (Sandevistan Mod config)
# 修改后保存，重启游戏生效；也可在游戏中按 F9 打开参数面板实时调节

# 触发热键（键码）：
#   220 = 反斜杠 \   33 = PageUp   34 = PageDown   36 = Home
#   188 = ,  190 = .  191 = /   112-123 = F1-F12
hotkey=220

# 斯安维斯坦生效时间（帧，30帧 = 1秒。默认 240 = 8秒）
duration=240

# 冷却时间（帧，默认 1200 = 40秒；0 = 无冷却）
cooldown=0

# 回放速度倍率（生效结束后快速回放玩家操作的倍率，默认 5）
replayspeed=5

# 残影生成间隔（帧，越小残影越密，默认 3）
ghostevery=3

# 时停期间粒子特效是否继续动画（1=继续 0=冻结，默认 1）
fxrun=1

# 残影混合模式（1=normal柔和 0=add发光，默认 1）——若背景破坏感明显可切换测试
ghostblend=1

# 残影不透明度（百分比 5-100，默认 25）——调低可减轻对场景的视觉干扰
ghostalpha=25

# 回放残影间隔（每 N 个历史帧生成 1 个残影，默认 6；调大可减少回放残影数量）
replayghost=12

# 启动时在左上角显示"模组已加载"标记（1=显示 0=隐藏，默认 1）
showmark=1

# 参数面板热键（默认 120 = F9；游戏内按此键打开模组参数面板）
panelkey=120

# 诊断日志（1=记录模组运行日志到 AppData，用于排查问题；默认 0）
diaglog=1

# 自动测试模式（仅供开发者测试，玩家保持 0）
debugtest=0
