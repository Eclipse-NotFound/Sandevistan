# 斯安维斯坦模组配置 (Sandevistan Mod config)
# 修改后保存，重启游戏生效；也可在游戏中按 F9 打开参数面板实时调节

# 触发热键（键码）：
#   220 = 反斜杠 \   33 = PageUp   34 = PageDown   36 = Home
#   188 = ,  190 = .  191 = /   112-123 = F1-F12
hotkey=220

# 斯安维斯坦生效时间（帧，30帧 = 1秒。默认 210 = 7秒）
duration=210

# 冷却时间（帧，默认 1200 = 40秒；0 = 无冷却）
cooldown=0

# 回放速度倍率（生效结束后快速回放玩家操作的倍率，默认 5）
replayspeed=5

# 残影生成间隔（帧，越小残影越密，默认 5）
ghostevery=5

# 时停期间粒子特效是否继续动画（1=继续 0=冻结，默认 1）
fxrun=1

# 残影混合模式（1=normal柔和 0=add发光，默认 1）——若背景破坏感明显可切换测试
ghostblend=1

# 残影不透明度（百分比 5-100，默认 70）
ghostalpha=70

# 回放残影生成间隔（每 N 个显示帧生成 1 个，默认 1；调大减少回放残影数量）
replayghost=1

# 回放残影寿命（显示帧，默认 4；越小残影消失越快，拖尾越短）
replayghostlife=4

# 投掷物可击落（1=开 0=关，默认 1）：被扔出的手雷、发射的导弹和榴弹
# 有血量，受击至 0 直接爆炸（不包括天角兽闪电等普通子弹）
projhits=1

# 投掷物血量（默认 30）
projhp=30

# 投掷物护甲（默认 0；预留接口——伤害先减护甲再扣血量）
projarmor=0

# ===== v1.114：按武器 id 单独设置血量/护甲（可选，不写=用上面的全局值）=====
# 键格式：projhp_<武器id> / projarmor_<武器id>
# 常用武器 id：grenade=手雷  bel=野火核弹  aglau=榴弹发射器  mlau=导弹发射器
# projhp_grenade=60
# projhp_bel=120
# projhp_aglau=45
# projarmor_aglau=10

# 启动时在左上角显示"模组已加载"标记（1=显示 0=隐藏，默认 1）
showmark=1

# 参数面板热键（默认 120 = F9；游戏内按此键打开模组参数面板）
panelkey=120

# 诊断日志（1=记录模组运行日志到 AppData，用于排查问题；默认 0）
diaglog=1

# 自动测试模式（仅供开发者测试，玩家保持 0）
debugtest=0

# 时停慢速倍率（1/N 速：5 = 敌人/物品 1/5 速度运动；默认 5）
slowfactor=5

# 残影配色模式（0=彩虹循环 1=边缘行者绿-蓝-紫渐变，按移动速度映射）
colormode=1

# 边缘行者渐变门槛（速度≥此值全绿；低于此值渐变为蓝紫）
edgethresh=15

# 疾跑中切枪（1=开 0=关，默认 1）：按住 Shift 疾跑时按数字键切换武器
# （游戏本体疾跑时数字键映射到第二组快捷槽，通常为空=无法切枪）
swaprun=1

# ===== v1.115：敌人斯安维斯坦（Enemy Sandevistan）=====
# 装备名单（逗号分隔的类名；默认：掠夺者/狮鹫(佣兵)/天角兽/英克雷/铁骑卫/斑马）
# 触发：敌人 AI 看到玩家进入战斗后立即开启；每房间装备名额 = 候选敌人数 × esandyper%
enemysandy=UnitRaider,UnitMerc,UnitAlicorn,UnitEncl,UnitRanger,UnitZebra
esandydur=150     # 敌人斯安维斯坦持续帧（30帧=1秒 → 5秒）
esandycd=300      # 冷却帧（10秒）
esandyspd=5       # 加速倍率
esandyghost=1     # 敌人残影（边缘行者配色）
esandymark=1      # 调试"S"徽标（绿=待机 黄=激活 灰=冷却）
esandyper=50      # v1.119 每房间装备百分比（50=一半 100=全部 0=关闭）
